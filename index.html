function uploadCSV(event) { const file = event.target.files[0]; if (!file) return; const cls = document.getElementById('filter-class').value; const sec = document.getElementById('filter-section').value; const expectedSubs = sectionConfigs[`${cls}-${sec}`] || []; const reader = new FileReader(); reader.onload = function(e) { const text = e.target.result.replace(/\r/g, ''); const lines = text.split('\n'); if (lines.length === 0 || !lines[0].trim()) { alert("❌ Upload Aborted: The uploaded CSV file is empty."); event.target.value = ''; return; } // 1. Strict Header Validation const headerCols = lines[0].split(',').map(c => c.replace(/^"|"$/g, '').trim().toUpperCase()); const uploadedSubs = headerCols.slice(3); const expectedClean = expectedSubs.map(s => s.trim().toUpperCase()); const uploadedClean = uploadedSubs.map(s => s.trim().toUpperCase()); let isHeaderMatch = expectedClean.length === uploadedClean.length && expectedClean.every((sub, i) => sub === uploadedClean[i]); if (!isHeaderMatch) { alert(`❌ CSV Subject Header Mismatch Error!\n\nExpected Headers for Class ${formatClass(cls)} - Sec ${sec}:\n${expectedClean.join(', ')}\n\nUploaded CSV Headers:\n${uploadedClean.join(', ')}\n\nUpload Aborted! Please use the official template.`); event.target.value = ''; return; } // 2. Strict Row Cell Validation (Checking that student subjects match master setup or are blank/'-') let parsedRows = []; for (let i = 1; i < lines.length; i++) { const line = lines[i].trim(); if(!line) continue; const cols = line.split(',').map(c => c.replace(/^"|"$/g, '').trim()); if (cols.length >= 3) { let studentMappedSubs = []; for (let subIdx = 0; subIdx < expectedClean.length; subIdx++) { let val = cols[3 + subIdx] ? cols[3 + subIdx].trim().toUpperCase() : '-'; // If cell is not empty, not '-', and doesn't match the master subject -> Trigger Mismatch Error! if (val && val !== '-' && val !== 'NIL' && val !== 'NONE') { if (val !== expectedClean[subIdx]) { alert(`❌ CSV Student Data Mismatch Error at Row ${i} (Roll: ${cols[1] || 'Unknown'})!\n\nInvalid subject entry found: "${val}" under column "${expectedClean[subIdx]}".\n\nStudents must write the exact master subject name or leave blank / '-' for unchosen subjects.\nUpload Aborted!`); event.target.value = ''; return; } studentMappedSubs.push(val); } else { studentMappedSubs.push('-'); } } parsedRows.push({ sno: parseInt(cols[0]) || (parsedRows.length + 1), roll: parseInt(cols[1]) || 0, name: cols[2], class: cls, section: sec, subjects: studentMappedSubs }); } } if (parsedRows.length === 0) { alert("⚠️ No valid student rows detected in the CSV file."); event.target.value = ''; return; } // 3. Push to Firebase if all rows pass validation let updates = {}; parsedRows.forEach(row => { const newKey = db.ref(`students/${cls}/${sec}`).push().key; row.id = newKey; updates[newKey] = row; }); db.ref(`students/${cls}/${sec}`).set(updates).then(() => { alert(`✅ CSV Uploaded & Synced Successfully! ${parsedRows.length} student records loaded.`); event.target.value = ''; switchTab('students'); }).catch((error) => { alert("❌ Firebase Sync Failed: " + error.message); event.target.value = ''; }); }; reader.onerror = function() { alert("❌ Error reading the uploaded file."); event.target.value = ''; }; reader.readAsText(file); }
